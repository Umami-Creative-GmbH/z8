/* @vitest-environment jsdom */
import {act,cleanup,fireEvent,render,screen,waitFor,within} from '@testing-library/react';
import {Activity,StrictMode} from 'react';
import {flushSync} from 'react-dom';
import {afterEach,beforeEach,describe,expect,it,vi} from 'vitest';
const mocks=vi.hoisted(()=>({start:vi.fn(),summary:vi.fn(),step:vi.fn(),push:vi.fn(),refetch:vi.fn(),session:{data:{user:{id:'a'},session:{activeOrganizationId:'org-a' as string|null}},isPending:false}}));
vi.mock('@tolgee/react',()=>({useTranslate:()=>({t:(_key:string,fallback:string,values?:Record<string,string>)=>values?fallback.replace('{organization}',values.organization):fallback})}));
vi.mock('@/lib/auth-client',()=>({useSession:()=>({...mocks.session,refetch:mocks.refetch})}));
vi.mock('@/navigation',()=>({useRouter:()=>({push:mocks.push})}));
vi.mock('@/app/[locale]/onboarding/welcome/actions',()=>({startOnboarding:mocks.start,getOnboardingSummary:mocks.summary,updateOnboardingStep:mocks.step}));
import WelcomePage from '@/app/[locale]/onboarding/welcome/page';
const joined={success:true,data:{hasOrganization:true,organizationName:'Current Org',isAdmin:true}};
const noOrg={success:true,data:{hasOrganization:false,organizationName:null,isAdmin:false}};
function deferred<T>(){let resolve!:(v:T)=>void;let reject!:(e:unknown)=>void;const promise=new Promise<T>((yes,no)=>{resolve=yes;reject=no});return{promise,resolve,reject}}
const page=(mode:'visible'|'hidden'='visible')=><Activity mode={mode}><main><WelcomePage/></main></Activity>;
const ready=()=>screen.findByRole('button',{name:'Continue Setup'});
const settle=()=>act(async()=>{await new Promise(r=>setTimeout(r,20))});
beforeEach(()=>{vi.resetAllMocks();mocks.session={data:{user:{id:'a'},session:{activeOrganizationId:'org-a'}},isPending:false};mocks.start.mockResolvedValue({success:true});mocks.summary.mockResolvedValue(joined);mocks.step.mockResolvedValue({success:true})});
afterEach(()=>cleanup());
describe('Independent assembled welcome public route',()=>{
 it.each(['session','initialization','summary'])('keeps one task heading and no continuation while %s is unavailable',async stage=>{if(stage==='session'){mocks.session.data.user.id='';mocks.session.isPending=true}else(stage==='initialization'?mocks.start:mocks.summary).mockReturnValue(new Promise(()=>{}));render(page());await settle();expect(within(screen.getByRole('main')).getAllByRole('heading',{level:1})).toHaveLength(1);expect(screen.queryByRole('button',{name:/Continue Setup|Get Started/})).toBeNull();expect(mocks.step).not.toHaveBeenCalled()});
 it('dispatches once in StrictMode and retains real shared progress',async()=>{render(<StrictMode>{page()}</StrictMode>);await ready();expect(mocks.start).toHaveBeenCalledOnce();expect(screen.getByRole('navigation',{name:'Onboarding progress'})).toBeTruthy()});
 it('keeps initialization successful when summary must retry',async()=>{mocks.summary.mockResolvedValueOnce({success:false,error:'Read refused'});render(page());fireEvent.click(await screen.findByRole('button',{name:'Retry setup'}));await ready();expect(mocks.start).toHaveBeenCalledOnce();expect(mocks.summary).toHaveBeenCalledTimes(2)});
 it('saves no-org destination only after a successful step result',async()=>{mocks.summary.mockResolvedValue(noOrg);const held=deferred<{success:true}>();mocks.step.mockReturnValueOnce(held.promise);render(page());fireEvent.click(await screen.findByRole('button',{name:'Get Started'}));expect(mocks.step).toHaveBeenCalledExactlyOnceWith('organization');expect(mocks.push).not.toHaveBeenCalled();await act(async()=>held.resolve({success:true}));expect(mocks.push).toHaveBeenCalledExactlyOnceWith('/onboarding/organization')});
 it('prevents duplicate same-turn clicks and locks during route continuation',async()=>{const held=deferred<{success:true}>();mocks.step.mockReturnValueOnce(held.promise);render(page());const button=await ready();act(()=>{fireEvent.click(button);fireEvent.click(button);fireEvent.click(button)});expect(mocks.step).toHaveBeenCalledOnce();await act(async()=>held.resolve({success:true}));fireEvent.click(button);expect(mocks.step).toHaveBeenCalledOnce();expect(mocks.push).toHaveBeenCalledOnce()});
 it.each(['account','organization'])('retires held success/refusal/rejection after separate %s A-B-A',async kind=>{for(const outcome of ['success','refused','rejected']){const held=deferred<unknown>();mocks.step.mockReturnValueOnce(held.promise);const view=render(page());fireEvent.click(await ready());mocks.session.data=kind==='account'?{user:{id:'b'},session:{activeOrganizationId:'org-a'}}:{user:{id:'a'},session:{activeOrganizationId:'org-b'}};view.rerender(page());await ready();mocks.session.data={user:{id:'a'},session:{activeOrganizationId:'org-a'}};view.rerender(page());await ready();await act(async()=>{if(outcome==='rejected')held.reject(new Error('private obsolete'));else held.resolve({success:outcome==='success',error:'obsolete refusal'})});expect(mocks.push).not.toHaveBeenCalled();expect(screen.queryByRole('alert')).toBeNull();view.unmount();mocks.push.mockClear()}});
 it('retires the old detached action after an account boundary',async()=>{const view=render(page());const old=await ready();mocks.session.data={user:{id:'b'},session:{activeOrganizationId:'org-b'}};flushSync(()=>view.rerender(page()));fireEvent.click(old);await settle();expect(mocks.step).not.toHaveBeenCalled();fireEvent.click(await ready());await waitFor(()=>expect(mocks.step).toHaveBeenCalledOnce())});
 it.each(['success','refused','rejected'])('ignores hidden Activity continuation %s and allows a fresh deliberate action',async outcome=>{const held=deferred<unknown>();mocks.step.mockReturnValueOnce(held.promise);const view=render(page());fireEvent.click(await ready());view.rerender(page('hidden'));await act(async()=>{if(outcome==='rejected')held.reject(new Error('private obsolete'));else held.resolve({success:outcome==='success',error:'obsolete refusal'})});expect(mocks.push).not.toHaveBeenCalled();view.rerender(page());await ready();expect(screen.queryByRole('alert')).toBeNull();fireEvent.click(await ready());await waitFor(()=>expect(mocks.push).toHaveBeenCalledOnce())});
 it('requires a refreshed summary before accepting an immediate Activity-return action',async()=>{const held=deferred<typeof noOrg>();mocks.summary.mockResolvedValueOnce(joined).mockReturnValueOnce(held.promise);const view=render(page());await ready();act(()=>{flushSync(()=>view.rerender(page('hidden')));flushSync(()=>view.rerender(page()));const stale=screen.queryByRole('button',{name:'Continue Setup'});if(stale)fireEvent.click(stale)});await settle();expect(mocks.step).not.toHaveBeenCalled();expect(mocks.push).not.toHaveBeenCalled();await act(async()=>held.resolve(noOrg));fireEvent.click(await screen.findByRole('button',{name:'Get Started'}));await waitFor(()=>expect(mocks.step).toHaveBeenExactlyOnceWith('organization'))});
 it('does not reissue an accepted initialization during ordinary parent rerender',async()=>{const view=render(page());await ready();view.rerender(page());await settle();expect(mocks.start).toHaveBeenCalledOnce();expect(mocks.summary).toHaveBeenCalledOnce()});
});
